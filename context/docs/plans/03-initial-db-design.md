```sql
CREATE TABLE IF NOT EXISTS users (
    id INTEGER NOT NULL UNIQUE GENERATED BY DEFAULT AS IDENTITY,
    user_uuid UUID NOT NULL UNIQUE,
    email VARCHAR(255) NOT NULL UNIQUE,
    phone_number VARCHAR(20) UNIQUE,
    auth_provider VARCHAR(20) NOT NULL DEFAULT 'email',
    auth_provider_id VARCHAR(100) UNIQUE,
    password_hash VARCHAR(255),
    last_login_role VARCHAR(50),
    first_name VARCHAR(255) NOT NULL,
    last_name VARCHAR(255) NOT NULL,
    email_verified BOOLEAN NOT NULL DEFAULT FALSE,
    phone_verified BOOLEAN NOT NULL DEFAULT FALSE,
    created_at TIMESTAMPTZ NOT NULL,
    updated_at TIMESTAMPTZ NOT NULL,
    PRIMARY KEY (id),
    CONSTRAINT ck_users_auth_provider CHECK (auth_provider IN ('email', 'google', 'github')),
    CONSTRAINT ck_users_last_login_role CHECK (
        last_login_role IS NULL OR last_login_role IN ('member', 'founder', 'reviewer', 'admin')
    )
);

CREATE INDEX users_idx_0 ON users (id, user_uuid, email);

CREATE TABLE IF NOT EXISTS roles (
    id INTEGER NOT NULL GENERATED BY DEFAULT AS IDENTITY,
    role_uuid UUID NOT NULL UNIQUE,
    name VARCHAR(255) NOT NULL,
    slug VARCHAR(50) NOT NULL UNIQUE,
    permissions JSONB NOT NULL,
    created_at TIMESTAMPTZ NOT NULL,
    updated_at TIMESTAMPTZ NOT NULL,
    PRIMARY KEY (id)
);

CREATE INDEX roles_idx_0 ON roles (id, role_uuid, slug);

CREATE TABLE IF NOT EXISTS profiles (
    id INTEGER NOT NULL UNIQUE GENERATED BY DEFAULT AS IDENTITY,
    profile_uuid UUID NOT NULL UNIQUE,
    user_id INTEGER NOT NULL UNIQUE,
    nick_name VARCHAR(100),
    role_id INTEGER NOT NULL,
    public_profile_url VARCHAR(255) NOT NULL,
    slug VARCHAR(50) NOT NULL UNIQUE,
    created_at TIMESTAMPTZ NOT NULL,
    updated_at TIMESTAMPTZ NOT NULL,
    PRIMARY KEY (id)
);

CREATE INDEX profiles_idx_0 ON profiles (id, profile_uuid, slug);

CREATE TABLE IF NOT EXISTS topics (
    id INTEGER NOT NULL GENERATED BY DEFAULT AS IDENTITY,
    topic_uuid UUID NOT NULL UNIQUE,
    text VARCHAR(255) NOT NULL,
    slug VARCHAR(100) NOT NULL UNIQUE,
    created_at TIMESTAMPTZ NOT NULL,
    updated_at TIMESTAMPTZ NOT NULL,
    PRIMARY KEY (id)
);

CREATE INDEX topics_idx_0 ON topics (id, topic_uuid, slug);

CREATE TABLE IF NOT EXISTS products (
    id INTEGER NOT NULL GENERATED BY DEFAULT AS IDENTITY,
    product_uuid UUID NOT NULL UNIQUE,
    profile_id INTEGER NOT NULL,
    name VARCHAR(255) NOT NULL,
    one_line_pitch VARCHAR(500) NOT NULL,
    description TEXT NOT NULL,
    logo_url VARCHAR(255) NOT NULL,
    screenshot_urls VARCHAR(255) ARRAY,
    website_url VARCHAR(255),
    status VARCHAR(20) NOT NULL DEFAULT 'draft',
    created_at TIMESTAMPTZ NOT NULL,
    updated_at TIMESTAMPTZ NOT NULL,
    PRIMARY KEY (id),
    CONSTRAINT ck_products_status CHECK (status IN ('draft', 'approved', 'scheduled', 'live', 'inactive'))
);

CREATE INDEX products_idx_0 ON products (id, product_uuid, profile_id);
CREATE INDEX products_idx_1 ON products (status, created_at);
CREATE INDEX products_idx_2 ON products (name);

CREATE TABLE IF NOT EXISTS product_topics (
    id INTEGER NOT NULL GENERATED BY DEFAULT AS IDENTITY,
    product_topic_uuid UUID NOT NULL UNIQUE,
    product_id INTEGER NOT NULL,
    topic_id INTEGER NOT NULL,
    created_at TIMESTAMPTZ NOT NULL,
    PRIMARY KEY (id),
    CONSTRAINT uq_product_topics_product_topic UNIQUE (product_id, topic_id)
);

CREATE INDEX product_topics_idx_0 ON product_topics (product_id, topic_id);

CREATE TABLE IF NOT EXISTS product_launches (
    id INTEGER NOT NULL GENERATED BY DEFAULT AS IDENTITY,
    product_launch_uuid UUID NOT NULL UNIQUE,
    product_id INTEGER NOT NULL UNIQUE,
    launch_date DATE NOT NULL,
    state VARCHAR(20) NOT NULL DEFAULT 'scheduled',
    launched_at TIMESTAMPTZ,
    snapshot_frozen_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL,
    updated_at TIMESTAMPTZ NOT NULL,
    PRIMARY KEY (id),
    CONSTRAINT ck_product_launches_state CHECK (state IN ('scheduled', 'live', 'archived', 'cancelled'))
);

CREATE INDEX product_launches_idx_0 ON product_launches (launch_date, state);

CREATE TABLE IF NOT EXISTS submissions (
    id INTEGER NOT NULL GENERATED BY DEFAULT AS IDENTITY,
    submission_uuid UUID NOT NULL UNIQUE,
    profile_id INTEGER NOT NULL,
    product_id INTEGER NOT NULL,
    status VARCHAR(20) NOT NULL DEFAULT 'draft',
    reviewed_by_profile_id INTEGER,
    review_note TEXT,
    reviewed_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL,
    updated_at TIMESTAMPTZ NOT NULL,
    PRIMARY KEY (id),
    CONSTRAINT ck_submissions_status CHECK (status IN ('draft', 'in_review', 'approved', 'rejected', 'changes_requested'))
);

CREATE INDEX submissions_idx_0 ON submissions (status, created_at);
CREATE INDEX submissions_idx_1 ON submissions (product_id, created_at);

CREATE TABLE IF NOT EXISTS votes (
    id INTEGER NOT NULL GENERATED BY DEFAULT AS IDENTITY,
    vote_uuid UUID NOT NULL UNIQUE,
    profile_id INTEGER NOT NULL,
    product_id INTEGER NOT NULL,
    created_at TIMESTAMPTZ NOT NULL,
    PRIMARY KEY (id),
    CONSTRAINT uq_votes_product_profile UNIQUE (product_id, profile_id)
);

CREATE INDEX votes_idx_0 ON votes (product_id, created_at);
CREATE INDEX votes_idx_1 ON votes (profile_id, created_at);

CREATE TABLE IF NOT EXISTS comments (
    id INTEGER NOT NULL GENERATED BY DEFAULT AS IDENTITY,
    comment_uuid UUID NOT NULL UNIQUE,
    profile_id INTEGER NOT NULL,
    product_id INTEGER NOT NULL,
    parent_comment_id INTEGER,
    text TEXT NOT NULL,
    created_at TIMESTAMPTZ NOT NULL,
    updated_at TIMESTAMPTZ NOT NULL,
    PRIMARY KEY (id)
);

CREATE INDEX comments_idx_0 ON comments (product_id, created_at);
CREATE INDEX comments_idx_1 ON comments (parent_comment_id, created_at);

CREATE TABLE IF NOT EXISTS reports (
    id INTEGER NOT NULL GENERATED BY DEFAULT AS IDENTITY,
    report_uuid UUID NOT NULL UNIQUE,
    reporter_profile_id INTEGER NOT NULL,
    target_type VARCHAR(20) NOT NULL,
    target_product_id INTEGER,
    target_comment_id INTEGER,
    reason VARCHAR(50) NOT NULL,
    details TEXT,
    status VARCHAR(20) NOT NULL DEFAULT 'open',
    assigned_reviewer_profile_id INTEGER,
    resolved_by_profile_id INTEGER,
    resolution_note TEXT,
    resolved_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL,
    updated_at TIMESTAMPTZ NOT NULL,
    PRIMARY KEY (id),
    CONSTRAINT ck_reports_target_type CHECK (target_type IN ('product', 'comment')),
    CONSTRAINT ck_reports_reason CHECK (
        reason IN ('spam', 'abuse', 'misleading', 'copyright', 'other')
    ),
    CONSTRAINT ck_reports_status CHECK (
        status IN ('open', 'in_review', 'resolved', 'dismissed')
    ),
    CONSTRAINT ck_reports_target_match CHECK (
        (target_type = 'product' AND target_product_id IS NOT NULL AND target_comment_id IS NULL)
        OR
        (target_type = 'comment' AND target_product_id IS NULL AND target_comment_id IS NOT NULL)
    )
);

CREATE INDEX reports_idx_0 ON reports (status, created_at);
CREATE INDEX reports_idx_1 ON reports (assigned_reviewer_profile_id, status);

CREATE TABLE IF NOT EXISTS notification_preferences (
    id INTEGER NOT NULL GENERATED BY DEFAULT AS IDENTITY,
    preference_uuid UUID NOT NULL UNIQUE,
    profile_id INTEGER NOT NULL UNIQUE,
    email_verification_enabled BOOLEAN NOT NULL DEFAULT TRUE,
    password_reset_enabled BOOLEAN NOT NULL DEFAULT TRUE,
    moderation_outcome_enabled BOOLEAN NOT NULL DEFAULT TRUE,
    launch_reminder_enabled BOOLEAN NOT NULL DEFAULT TRUE,
    product_live_enabled BOOLEAN NOT NULL DEFAULT TRUE,
    comment_alert_enabled BOOLEAN NOT NULL DEFAULT TRUE,
    daily_top_five_enabled BOOLEAN NOT NULL DEFAULT TRUE,
    weekly_best_enabled BOOLEAN NOT NULL DEFAULT TRUE,
    created_at TIMESTAMPTZ NOT NULL,
    updated_at TIMESTAMPTZ NOT NULL,
    PRIMARY KEY (id)
);

CREATE TABLE IF NOT EXISTS notifications (
    id INTEGER NOT NULL GENERATED BY DEFAULT AS IDENTITY,
    notification_uuid UUID NOT NULL UNIQUE,
    recipient_profile_id INTEGER NOT NULL,
    actor_profile_id INTEGER,
    event_type VARCHAR(50) NOT NULL,
    payload JSONB NOT NULL,
    read_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL,
    PRIMARY KEY (id)
);

CREATE INDEX notifications_idx_0 ON notifications (recipient_profile_id, created_at);
CREATE INDEX notifications_idx_1 ON notifications (recipient_profile_id, read_at);

CREATE TABLE IF NOT EXISTS notification_deliveries (
    id INTEGER NOT NULL GENERATED BY DEFAULT AS IDENTITY,
    delivery_uuid UUID NOT NULL UNIQUE,
    notification_id INTEGER NOT NULL,
    channel VARCHAR(20) NOT NULL,
    provider VARCHAR(50) NOT NULL,
    provider_message_id VARCHAR(255),
    attempt_count INTEGER NOT NULL DEFAULT 0,
    status VARCHAR(20) NOT NULL DEFAULT 'pending',
    next_retry_at TIMESTAMPTZ,
    last_error TEXT,
    delivered_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL,
    updated_at TIMESTAMPTZ NOT NULL,
    PRIMARY KEY (id),
    CONSTRAINT ck_notification_deliveries_channel CHECK (channel IN ('email', 'in_app')),
    CONSTRAINT ck_notification_deliveries_status CHECK (
        status IN ('pending', 'retrying', 'sent', 'failed', 'dropped')
    )
);

CREATE INDEX notification_deliveries_idx_0 ON notification_deliveries (status, next_retry_at);
CREATE INDEX notification_deliveries_idx_1 ON notification_deliveries (notification_id, status);

CREATE TABLE IF NOT EXISTS staff_actions (
    id INTEGER NOT NULL GENERATED BY DEFAULT AS IDENTITY,
    staff_action_uuid UUID NOT NULL UNIQUE,
    actor_profile_id INTEGER NOT NULL,
    action_type VARCHAR(50) NOT NULL,
    target_type VARCHAR(50) NOT NULL,
    target_id INTEGER NOT NULL,
    reason TEXT,
    metadata JSONB NOT NULL DEFAULT '{}'::jsonb,
    created_at TIMESTAMPTZ NOT NULL,
    PRIMARY KEY (id)
);

CREATE INDEX staff_actions_idx_0 ON staff_actions (actor_profile_id, created_at);
CREATE INDEX staff_actions_idx_1 ON staff_actions (target_type, target_id, created_at);

CREATE TABLE IF NOT EXISTS audit_logs (
    id INTEGER NOT NULL GENERATED BY DEFAULT AS IDENTITY,
    audit_log_uuid UUID NOT NULL UNIQUE,
    actor_profile_id INTEGER,
    event_type VARCHAR(100) NOT NULL,
    entity_type VARCHAR(50) NOT NULL,
    entity_id INTEGER,
    before_state JSONB,
    after_state JSONB,
    correlation_id VARCHAR(100),
    ip_address VARCHAR(64),
    user_agent TEXT,
    created_at TIMESTAMPTZ NOT NULL,
    PRIMARY KEY (id)
);

CREATE INDEX audit_logs_idx_0 ON audit_logs (event_type, created_at);
CREATE INDEX audit_logs_idx_1 ON audit_logs (entity_type, entity_id, created_at);
CREATE INDEX audit_logs_idx_2 ON audit_logs (actor_profile_id, created_at);

CREATE TABLE IF NOT EXISTS member_follows (
    id INTEGER NOT NULL GENERATED BY DEFAULT AS IDENTITY,
    member_follow_uuid UUID NOT NULL UNIQUE,
    follower_profile_id INTEGER NOT NULL,
    followed_profile_id INTEGER NOT NULL,
    created_at TIMESTAMPTZ NOT NULL,
    PRIMARY KEY (id),
    CONSTRAINT uq_member_follows_pair UNIQUE (follower_profile_id, followed_profile_id),
    CONSTRAINT ck_member_follows_not_self CHECK (follower_profile_id <> followed_profile_id)
);

CREATE INDEX member_follows_idx_0 ON member_follows (follower_profile_id, created_at);
CREATE INDEX member_follows_idx_1 ON member_follows (followed_profile_id, created_at);

CREATE TABLE IF NOT EXISTS topic_follows (
    id INTEGER NOT NULL GENERATED BY DEFAULT AS IDENTITY,
    topic_follow_uuid UUID NOT NULL UNIQUE,
    profile_id INTEGER NOT NULL,
    topic_id INTEGER NOT NULL,
    created_at TIMESTAMPTZ NOT NULL,
    PRIMARY KEY (id),
    CONSTRAINT uq_topic_follows_pair UNIQUE (profile_id, topic_id)
);

CREATE INDEX topic_follows_idx_0 ON topic_follows (profile_id, created_at);
CREATE INDEX topic_follows_idx_1 ON topic_follows (topic_id, created_at);
```
